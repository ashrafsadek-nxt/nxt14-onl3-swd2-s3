src/axiosClient.js
==================
import axios from "axios";

const axiosClient = axios.create({
  baseURL: "https://jsonplaceholder.typicode.com",
  timeout: 10000, // 10s timeout
});

axiosClient.interceptors.request.use(
  (config) => {
    console.log("[axios] Request:", config.method?.toUpperCase(), config.url, config.params || config.data || "");
    return config;
  },
  (error) => {
    console.error("[axios] Request error:", error);
    return Promise.reject(error);
  }
);

axiosClient.interceptors.response.use(
  (response) => {
    console.log("[axios] Response:", response.status, response.config.url);
    return response;
  },
  (error) => {
    if (axios.isCancel(error)) {
      console.log("[axios] Request canceled:", error.message);
      return Promise.reject({ name: "CanceledError", message: error.message });
    }

    if (error.response) {
      const { status, data } = error.response;
      console.error("[axios] Response error:", status, data);
      return Promise.reject({ name: "HTTPError", message: `HTTP ${status}`, status, data });
    } else if (error.request) {
      console.error("[axios] No response:", error.message);
      return Promise.reject({ name: "NoResponse", message: error.message });
    } else {

      console.error("[axios] Setup error:", error.message);
      return Promise.reject({ name: "SetupError", message: error.message });
    }
  }
);

export default axiosClient;

import React, { useState, useEffect } from "react";
import axiosClient from "./axiosClient";

export default function UserProfile({ userId }) {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);

  useEffect(() => {
    console.log("useEffect triggered for userId =", userId);

    if (!userId) {
      setUser(null);
      setLoading(false);
      setError(null);
      return;
    }

    const controller = new AbortController();
    const signal = controller.signal;
    let mounted = true;

    setLoading(true);
    setError(null);

    (async () => {
      try {
        const res = await axiosClient.get(`/users/${userId}`, { signal });

        if (!mounted) return; 
        console.log("Fetched user (axios):", res.data);
        setUser(res.data);
      } catch (err) {
        console.error("Axios fetch error:", err);

        if (err.name === "CanceledError" || err.message === "canceled") {
          console.log("Request aborted for userId =", userId);
        } else if (err.name === "HTTPError" || err.name === "NoResponse" || err.name === "SetupError") {
          if (mounted) setError(err.message || "Unknown axios error");
        } else {
          if (mounted) setError(err.message || "Unknown error");
        }
      } finally {
        if (mounted) setLoading(false);
      }
    })();

    return () => {
      mounted = false;
      controller.abort(); 
      console.log("Cleanup for userId =", userId);
    };
  }, [userId]);

  if (!userId) return <div>choose user</div>;
  if (loading) return <div>loading...</div>;
  if (error) return <div>error: {error}</div>;
  if (!user) return <div>no data</div>;

  return (
    <div style={{ marginTop: 12 }}>
      <h3>{user.name}</h3>
      <p><strong>Email:</strong> {user.email}</p>
      <p><strong>Phone:</strong> {user.phone}</p>
      <pre style={{ background: "#f5f5f5", padding: 8 }}>
        {JSON.stringify(user, null, 2)}
      </pre>
    </div>
  );
}
